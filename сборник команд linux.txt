ls - вывести файлы в каталоге
    -a --all			выводит скрытые файлы
    -l				файлы идут в столбик(т.е по нормальному)
    -t				сначала новые
    -f				в алфавитном порядке
    -d --directory "папка"	покажет существует ли папка, вместо содержимого
    -h --human-readable		нормально отображает размеры файлов
    
rm - удалить файл/папку
    -f		удалить принудительно
    -i		спрашивать разрешение на удаление на каждый файл
    -I		спрашивать разрешение на удаление 1 раз если удалять больше 3 файлов или рекурсивно
    -r -R	удалять рекурсивно, т.е папки и их содержимое
    -d		удалить папку только если она пуста

touch - создать файл или изменить время создания/изменения
    -t YYYYMMDDhhmm	устанавливает время создания и изменения файлу
    -c --no-create	если файла нет то не создавать новый
    -a			только дату создания
    -m			только дату изменения
    -r другой_файл	устанавливает время создания и изменения как у другого файла

cp "папка/файл" "другая папка"- скопировать папку/файл в другую папку
    -r --recursive	скопирут папку рекурсивно(со всем внутри неё)
    -f --force		насильно скопировать

cat - вывести содержимое файла в консоль
    -A --show-all	   выводит скрытые символы(Переносы строки, TAB и др)
    -n --number		   выводит номера строк
    -s --squeeze-blank	   убирает пустые строки, идущие подряд
    
less - как cat но не загружает весь файл сразу(помогает если файл огромный)

sort [файл] - отсортировать текст из файла/конвейера 	
    -h 		сортировать по нормальным величинам(1М > 100К)
    -M		сортировать по аббревиатурам месяцев
    -n		сортировать по "числовому значению строки" (дословный перевод)
    -g		сортировать по "общему числовому значению строки" (тоже дословный перевод)
    -V		сортировать по версии SemVer (1.12.2 > 1.1.2)
    -d		сортировать в алфавитном порядке
    -m  	объединять уже отсортированные файлы, не сортировать (дословный перевод)
    -f		игнорировать регистр
    -i		игнорировать символы типа табуляции, переноса строки и тд
    -b		игнорировать пробелы в начале строк
    -r		инверсировать результат
    -u		убрать повторы
    -z		NUL - разделитель строк вместо переноса строки
   --parallel X изменить кол-во параллельных потоков на X

chmod [к кому применяется]+-="права" "файл/папка" - изменить права доступа на файл
    права:
    0 или -	ничего
    1 или x	исполнение ИЛИ переход в папку
    2 или w	запись
    4 или r	чтение
    
    к кому применяется:
    u - владелец файла
    g - группа файла
    o - остальные юзеры
    a - вообще все(перезаписывает все свыше)
    
    файл/папка:
    -R --recursive	рекурсивно применит права ко всем файлам в папке
					(без -R изменит ТОЛЬКО права на саму папку и файлы в ней не затронет)		
    
    --reference=RFILE	скопирует права с RFILE и применит их на файл/папку 

grep "слово" - найти строку с словом
    -i --ignore-case		игнорирует регистр
    -v --invert-match		инвертирует результат
    -c --count			количество подходящих строк
    -r --recursive		искать во всех файлах в папке
    -n --line-number		выводит номера подходящих строк
    -l --files-with-matches	выводит имена файлов в которых были найдены строки
    -L --files-without-match	выводит имена файлов в которых не были найдены строки
    -E --extended-regexp	спец символы типа + | ? {} воспринимаются как спец символы(без -E перед ними нужен \)
    -F --fixed-strings		любой символ воспринимается буквально
    -- "-слово"			помогает если grep воспринимает слово как аргумент
    
    "^слово"	выведет все строки которые начинаются на слово
    "слово$"	выведет все строки которые кончаются на слово
    
    "\<слово"	всё что начинается на слово
    "слово\>"	всё что кончается на слово    
    
head - напечатать первые 10 строк файла(ов)
    -n --lines число	напечатать первые "число" строк
    -c --bytes число	напечатать первые "число" байт
    -q --quiet		не печатать строки обозначающие файлы из которых взял строку
    -v --verbose 	печатать строки обозначающие файлы из которых взял строку

tail - напечатать последние 10 строк файла(ов)
    -n --lines число	напечатать последние "число" строк
    -c --bytes число	напечатать последние "число" байт
    -f --follow		печатать последние строки в реальном времени, а не статично
    -q --quiet		не печатать строки, обозначающие файлы из которых взял строку
    -v --verbose 	печатать строки, обозначающие файлы из которых взял строку

vim - ублюдский текстовый редактор, но по словам ИИ появляется чаще чем nano, так что приходится писать
    ESC - Normal mode(в нём пишутся команды)
    I - Insert mode(в нём пишется текст)
    Shift+R - Replace mode(как Insert но пишешь поверх текста)
    
    команды:
    :w - записать в файл
    :q - выйти если не было изменений
    :wq - выйти и сохранить изменения
    :qa! - выйти принудительно без сохранения
    /текст + Enter - найти текст(N на следущее совпадение, Shift+N на предыдущее)
    u - отменить последнее действие(как Ctrl+Z)
    Ctrl + r - вернуть последнее действие(как Ctrl+Y или Ctrl+Shift+Z)
    dd - удалить всю строку
    yy - скопировать всю строку
    p - вставить

iptables "откуда брать пакеты" "действие с ними " [опции] - управление брандмауэром(файрволом)
    
    -A/D/I.. источники пакетов
    INPUT - на само устройство
    OUTPUT - то что отправляет само устройство
    FORWARD - отдать другому устройству
    PREROUTING - обработка пакета до INPUT'ов и прочих
    POSTROUTING - обработка пакета после всех INPUT'ов и тд
    
    -j действия:
    ACCEPT - принять запрос
    DROP - молча скинуть запрос будто ничего сервера и нет
    REJECT - скинуть, но с сообщением об ошибке
    LOG - записать в логи информацию о запросе
    
    опции:
    -p --protocol 			протокол(tcp/udp) 
    -s --source-address/port/mask	с какого конкретно адреса/порта/маски
    -d --destination-address/port/mask	в какой конкретно адрес/порт/маску
    
    -i --in-interface интерфейс		с какого конкретно интерфейса*
    -o --out-interface интерфейс	в какой конкретно интерфейс*
    
    
    
    
    
    
    
    *интерфейсы
    eth0 - трафик извне
    eth1 - трафик внутри локалки
    lo - трафик идущий через 127.0.0.1 т.е localhost
